        workerUrl: WORKER_URL,
        token:     WORKER_TOKEN,
        storeId:   localStorage.getItem('lb_store_id') || '',
      }, '*');

    return;
  }

  // Notificação comum
  if (d.type === 'NOTIF') {
    const notif = {
      from: d.from || 'Sistema',
      txt:  d.txt  || '',
      tab:  d.tab  || '',
      ts:   Date.now(),

  }
});

function refreshLiveTiles() {
  // Tarefas
  try {
    const tarefas = JSON.parse(localStorage.getItem('hiper_tarefas') || '[]');
    const pend = tarefas.filter(t => !t.feita).length;

    }
  } catch(e) {}

  // Feed
  try {
    const posts = JSON.parse(localStorage.getItem('hiper_posts') || '[]');
    const feedSub = document.getElementById('tile-feed-count');
    if (feedSub) feedSub.textContent = posts.length + ' posts';
    const badgeFeed = document.getElementById('badge-feed');

    const recentes = posts.filter(p => p.ts && Date.now()-p.ts < 7200000).length;
    if (badgeFeed) { badgeFeed.textContent = recentes; badgeFeed.classList.toggle('show', recentes > 0); }
  } catch(e) {}

  // Agenda — eventos hoje
  try {
    const hoje = new Date();
    const chave = `${hoje.getFullYear()}-${String(hoje.getMonth()+1).padStart(2,'0')}-${String(hoje.getDate()).padStart(2,'0')}`;

let _pollingTimer = null;
let _pedidosIds = new Set();

async function fetchWorkerTile() {
  if (!WORKER_URL) return;
  const dot = document.getElementById('worker-dot');
  try {
    const r = await fetch(`${WORKER_URL}/pedidos?status=pendente`, {
      headers: { 'X-LABX-Token': WORKER_TOKEN },
      signal: AbortSignal.timeout(8000)
    });
    if (!r.ok) throw new Error('not ok');
    const d = await r.json();
    const pedidos = d.pedidos || [];
    const total   = pedidos.reduce((s, p) => s + (p.total || 0), 0);

    const countEl = document.getElementById('tile-pedidos-count');
    const totalEl = document.getElementById('tile-pedidos-total');
    if (countEl) countEl.textContent = pedidos.length;
    if (totalEl) totalEl.textContent = 'R$ ' + total.toFixed(2).replace('.', ',');

    const badge = document.getElementById('badge-labsystem');
    if (badge) { badge.textContent = pedidos.length; badge.classList.toggle('show', pedidos.length > 0); }

    // Novos pedidos → beep + notif
    const novos = pedidos.filter(p => !_pedidosIds.has(p.id));
    if (novos.length && _pedidosIds.size > 0) {
      _beep(880, 0.3);
      showToast(`📦 ${novos.length} novo(s) pedido(s)!`);
      novos.forEach(p => {
        _notifs.unshift({ from:'📊 LabSystem', txt:`Novo pedido #${p.id} · R$ ${(p.total||0).toFixed(2)}`, ts: Date.now(), tab:'labsystem' });
      });
      localStorage.setItem('lb_notifs', JSON.stringify(_notifs));
      updateNotifBadge();
    }
    novos.forEach(p => _pedidosIds.add(p.id));

    // SELIC tile
    fetchFeed();

    dot.classList.add('online');
    dot.classList.remove('offline');
  } catch(e) {
    dot.classList.remove('online');
    dot.classList.add('offline');
  }
}

async function fetchFeed() {
  if (!WORKER_URL) return;
  try {
    const r = await fetch(`${WORKER_URL}/feed`, {
      headers: { 'X-LABX-Token': WORKER_TOKEN },
      signal: AbortSignal.timeout(6000)
    });
    const d = await r.json();
    const el = document.getElementById('tile-selic');
    if (el && d.selic) el.textContent = `SELIC ${d.selic.valor}%`;
  } catch(e) {}
}

function _beep(freq, vol) {
  try {
    const ctx = new (window.AudioContext || window.webkitAudioContext)();
    const osc = ctx.createOscillator();
    const g   = ctx.createGain();
    osc.connect(g); g.connect(ctx.destination);
    osc.frequency.value = freq;
    g.gain.setValueAtTime(vol, ctx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.3);
    osc.start(); osc.stop(ctx.currentTime + 0.3);
  } catch(e) {}
}

function startPolling() {
  if (_pollingTimer) clearInterval(_pollingTimer);
  fetchWorkerTile();
  _pollingTimer = setInterval(fetchWorkerTile, 30000);
}

'use strict';

window.LBNotif = (function() {

  /* ── STATE ──────────────────────────────────────────────────── */
  const S = {
    items:      [],
    unread:     0,
    panelOpen:  false,
    soundOn:    true,
  };

  /* ── TIME ───────────────────────────────────────────────────── */
  function now() {
    const d = new Date();
    return String(d.getHours()).padStart(2,'0') + ':' + String(d.getMinutes()).padStart(2,'0');
  }

  /* ── ADD NOTIFICATION ───────────────────────────────────────── */
  function add(from, txt, type = 'info') {
    const n = { id: Date.now() + Math.random(), from, txt, type, time: now() };
    S.items.unshift(n);
    if (S.items.length > 60) S.items.length = 60;
    S.unread++;
    _updateBadge();
    _renderList();
    toast(txt.slice(0, 64), type);
    if (S.soundOn && (type === 'order' || type === 'success')) beep();
    return n.id;
  }

  /* ── DISMISS ────────────────────────────────────────────────── */
  function dismiss(id) {
    S.items = S.items.filter(n => n.id !== id);
    _renderList();
  }

  /* ── CLEAR ALL ──────────────────────────────────────────────── */
  function clearAll() {
    S.items  = [];
    S.unread = 0;
    _updateBadge();
    _renderList();
  }

  /* ── BADGE ──────────────────────────────────────────────────── */
  function _updateBadge() {
    const el = document.getElementById('lb-notif-cnt');
    if (!el) return;
    if (S.unread > 0) {
      el.textContent = S.unread > 9 ? '9+' : S.unread;
      el.classList.add('show');
    } else {
      el.classList.remove('show');
    }
    // Taskbar button alert indicator
    const tbBtn = document.getElementById('tb-notif-btn');
    if (tbBtn) tbBtn.classList.toggle('has-alert', S.unread > 0);
  }

  /* ── RENDER LIST ────────────────────────────────────────────── */
  function _renderList() {
    const list = document.getElementById('lb-notif-list');
    if (!list) return;
    if (!S.items.length) {
      list.innerHTML = '<div class="np-empty">Sem notificações</div>';
      return;
    }
    const ICONS = { info:'📌', success:'✅', error:'❌', warn:'⚠️', order:'📦' };
    list.innerHTML = S.items.map(n => `
      <div class="np-item type-${n.type}" id="np-${n.id}">
        <span class="np-icon">${ICONS[n.type]||'📌'}</span>
        <div class="np-body">
          <div class="np-from">${_esc(n.from)}</div>
          <div class="np-txt">${_esc(n.txt)}</div>
          <div class="np-time">${n.time}</div>
        </div>
        <button class="np-dismiss" onclick="LBNotif.dismiss(${n.id})">✕</button>
      </div>
    `).join('');
  }

  /* ── TOGGLE PANEL ───────────────────────────────────────────── */
  function togglePanel() {
    const panel = document.getElementById('lb-notif-panel');
    if (!panel) return;
    S.panelOpen = !S.panelOpen;
    panel.classList.toggle('open', S.panelOpen);
    if (S.panelOpen) {
      S.unread = 0;
      _updateBadge();
      _renderList();
      // Fecha ao clicar fora
      setTimeout(() => {
        document.addEventListener('click', _outsideClose, { once: true });
      }, 50);
    }
  }
  window.closeNotifPanel = function() {
    S.panelOpen = false;
    document.getElementById('lb-notif-panel')?.classList.remove('open');
  };

  function _outsideClose(e) {
    const p = document.getElementById('lb-notif-panel');
    const b = document.getElementById('lb-notif-bell');
    if (p && !p.contains(e.target) && e.target !== b && !b?.contains(e.target)) {
      closeNotifPanel();
    }
  }

  /* ── TOAST ──────────────────────────────────────────────────── */
  function toast(msg, type = 'info', ms = 3400) {
    const area = document.getElementById('lb-toasts');
    if (!area) return;
    const ICONS = { info:'ℹ️', success:'✅', error:'❌', warn:'⚠️', order:'📦' };
    const el = document.createElement('div');
    el.className = `lb-toast type-${type}`;
    el.innerHTML = `<span class="lb-toast-icon">${ICONS[type]||'ℹ️'}</span><span>${_esc(msg)}</span>`;
    area.prepend(el);
    setTimeout(() => {
      el.classList.add('out');
      setTimeout(() => el.remove(), 220);
    }, ms);
  }

  /* ── BEEP ───────────────────────────────────────────────────── */
  function beep(freq = 880, dur = 0.32, vol = 0.3) {
    if (!S.soundOn) return;
    try {
      const ctx = new (window.AudioContext || window.webkitAudioContext)();
      const osc = ctx.createOscillator();
      const g   = ctx.createGain();
      osc.connect(g); g.connect(ctx.destination);
      osc.type = 'sine';
      osc.frequency.setValueAtTime(freq, ctx.currentTime);
      g.gain.setValueAtTime(vol, ctx.currentTime);
      g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + dur);
      osc.start(); osc.stop(ctx.currentTime + dur);
    } catch(e) {}
  }

  /* ── TOGGLE SOUND ───────────────────────────────────────────── */
  function toggleSound() {
    S.soundOn = !S.soundOn;
    const btn = document.getElementById('qa-sound');
    if (btn) btn.classList.toggle('on', S.soundOn);
    toast(S.soundOn ? 'Sons ativados' : 'Sons desativados', 'info', 2000);
  }

  /* ── postMessage BRIDGE (iframes → shell) ───────────────────── */
  function initBridge() {
    window.addEventListener('message', e => {
      const d = e.data;
      if (!d || typeof d !== 'object') return;
      switch (d.type) {
        case 'NOTIF':
          add(d.from || 'Lab', d.txt || '', d.notifType || 'info');
          break;
        case 'POSTIT':
          add('🗒 ' + (d.from || 'Post-it'), d.txt || '', 'warn');
          break;
        case 'ORDER_NEW':
          add('📦 Novo Pedido', d.txt || 'Pedido recebido', 'order');
          break;
        case 'TOAST':
          toast(d.msg || '', d.toastType || 'info');
          break;
      }
    });
  }

  /* ── ESCAPE ─────────────────────────────────────────────────── */
  function _esc(str) {
    return String(str)
      .replace(/&/g,'&amp;').replace(/</g,'&lt;')
      .replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  /* ── INIT ───────────────────────────────────────────────────── */
  function init() {
    _renderList();
    initBridge();
    // Botão sino
    document.getElementById('lb-notif-bell')?.addEventListener('click', togglePanel);
    // Botão taskbar
    document.getElementById('tb-notif-btn')?.addEventListener('click', () => {
      togglePanel();
      document.getElementById('tb-notif-btn')?.classList.remove('has-alert');
    });
    // Quick Actions
    document.getElementById('qa-sound')?.addEventListener('click', toggleSound);
    document.getElementById('qa-clear')?.addEventListener('click', clearAll);
    document.getElementById('qa-search')?.addEventListener('click', () => {
      closeNotifPanel(); window.openSearch?.();
    });
    document.getElementById('qa-orders')?.addEventListener('click', () => {
      closeNotifPanel(); window.openWindow?.('labsystem');
    });
    // Boas-vindas
    setTimeout(() => toast("la'browser iniciado", 'success', 2600), 500);
  }

  /* ── PUBLIC API ─────────────────────────────────────────────── */
  return { add, dismiss, clearAll, toast, beep, togglePanel, toggleSound, init };

})();

function $t(id, val) {

  return el;

/* ── WORKER POLLING ─────────────────────────────────────────────── */
async function pollWorker() {
  const dot = document.getElementById('lb-worker-dot');
  try {
    const r = await fetch(`${LB_WORKER.url}/pedidos?status=pendente`, {
      headers: { 'X-LABX-Token': LB_WORKER.token }
    });
    if (!r.ok) throw new Error(r.status);
    const d = await r.json();
    const pedidos = d.pedidos || [];
    const total   = pedidos.reduce((s, p) => s + (p.total || 0), 0);

    /* Tile LabSystem */
    $t('tile-ped-count', pedidos.length);
    $t('tile-ped-total', 'R$ ' + total.toFixed(2).replace('.', ','));
    $badge('badge-labsystem', pedidos.length);

    /* Taskbar alert */
    document.querySelector('[data-pinid="labsystem"]')?.classList.toggle('has-alert', pedidos.length > 0);

    /* Detecta novos */
    const novos = pedidos.filter(p => !LB_WORKER._known.has(p.id));
    if (novos.length && LB_WORKER._known.size > 0) {
      const EMOJ = { vitorelli:'🍕', pontox:'🥟', moraes:'🔥', totem:'🖥' };
      novos.forEach(p => {
        const e = EMOJ[p.lojaId] || '📦';
        window.LBNotif?.add(
          `${e} Pedido · ${p.lojaId || '—'}`,
          `R$ ${(p.total||0).toFixed(2)} · ${p.cliente || 'Cliente'}`,
          'order'
        );
      });
      window.LBNotif?.beep(880);
    }
    novos.forEach(p => LB_WORKER._known.add(p.id));

    /* Worker online */
    if (dot) { dot.classList.add('online'); dot.classList.remove('offline'); }

    /* SELIC */
    _pollSelic();

  } catch(e) {
    if (dot) { dot.classList.remove('online'); dot.classList.add('offline'); }
  }
}

async function _pollSelic() {
  try {
    const r = await fetch(`${LB_WORKER.url}/feed`);
    if (!r.ok) return;
    const d = await r.json();
    if (d.selic?.valor) $t('tile-selic', 'SELIC ' + parseFloat(d.selic.valor).toFixed(2) + '%');
    if (d.ipca?.valor)  $t('tile-ipca',  'IPCA '  + parseFloat(d.ipca.valor).toFixed(2)  + '%');
  } catch(e) {}
}

/* ── LOCAL DATA TILES (HiperApp localStorage) ───────────────────── */
function refreshLocalTiles() {
  /* Tarefas */
  try {
    const ts = JSON.parse(localStorage.getItem('hiper_tarefas') || '[]');
    const pend = ts.filter(t => !t.feita).length;
    const alta = ts.filter(t => !t.feita && t.pri === 'alta').length;
    $t('tile-tar-count', pend);
    $badge('badge-tarefas', alta);
  } catch(e) {}

  /* Última nota */
  try {
    const diarios = JSON.parse(localStorage.getItem('hiper_diarios') || '{}');
    const keys = Object.keys(diarios);
    if (keys.length) {
      const ult = diarios[keys[keys.length - 1]];
      const pg0 = (ult?.pages || [[]])[0] || [];
      const txt = pg0.find(b => b?.text)?.text || 'Diário de Bordo';
      $t('tile-nota-preview', txt.slice(0, 30) + (txt.length > 30 ? '…' : ''));
      if (ult?.criadoEm) $t('tile-nota-data', new Date(ult.criadoEm).toLocaleDateString('pt-BR'));
    }
  } catch(e) {}

  /* Feed — último post */
  try {
    const posts = JSON.parse(localStorage.getItem('hiper_posts') || '[]');
    if (posts[0]) $t('tile-feed-last', (posts[0].txt || '').slice(0, 28) + '…');
  } catch(e) {}

  /* Agenda — eventos hoje */
  try {
    const evs = JSON.parse(localStorage.getItem('hiper_eventos') || '{}');
    const hj  = new Date();
    const k   = `${hj.getFullYear()}-${String(hj.getMonth()+1).padStart(2,'0')}-${String(hj.getDate()).padStart(2,'0')}`;
    const hoje = evs[k] || [];
    if (hoje.length) {
      $t('tile-agenda-next', hoje[0].txt?.slice(0, 22) || 'Evento');
      $t('tile-agenda-sub', hoje.length + ' evento(s) hoje');
    }
  } catch(e) {}

  /* Figurinhas */
  try {
    const alb = JSON.parse(localStorage.getItem('mimo_album') || '[]');
    $t('tile-figurinhas', alb.filter(Boolean).length + ' figurinhas');
  } catch(e) {}
}

window.openSearch = function() {
  const ov = document.getElementById('lb-search-overlay');
  if (!ov) return;
  ov.classList.add('open');
  setTimeout(() => {
    const inp = document.getElementById('lb-search-inp');
    if (inp) { inp.value = ''; inp.focus(); doSearch(''); }
  }, 80);
};

  document.getElementById('lb-search-overlay')?.classList.remove('open');
};

window.doSearch = function(q) {
  const res = document.getElementById('lb-search-results');
  if (!res) return;
  const lista = q.trim()
    ? LB_LABS.filter(l =>
        l.nome.toLowerCase().includes(q.toLowerCase()) ||
        l.desc.toLowerCase().includes(q.toLowerCase()))
    : LB_LABS;
  if (!lista.length) {
    res.innerHTML = '<div style="color:rgba(255,255,255,.28);padding:20px;font-size:10px;letter-spacing:1.5px;text-transform:uppercase;">Sem resultados</div>';
    return;
  }
  res.innerHTML = lista.map(l => `
    <div class="sr-item" onclick="closeSearch();openWindow('${l.id}')">
      <div class="sr-icon">${l.icon}</div>
      <div>
        <div class="sr-name">${l.nome}</div>
        <div class="sr-desc">${l.desc}</div>
      </div>
    </div>
  `).join('');
};

function _bindOmniDrag(el) {
  const bar = el.querySelector('.lb-win-bar');
  let ox = 0, oy = 0, on = false;
  bar.onmousedown = e => {
    if (e.target.closest('.lb-win-controls')) return;
    on = true; ox = e.clientX - el.offsetLeft; oy = e.clientY - el.offsetTop;
    document.onmousemove = e => {
      if (!on) return;
      el.style.left = Math.max(0, e.clientX - ox) + 'px';
      el.style.top  = Math.max(28, e.clientY - oy) + 'px';
    };
    document.onmouseup = () => { on = false; document.onmousemove = null; document.onmouseup = null; };
  };
}

  h.onmousedown = e => {
    on = true; sx = e.clientX; sy = e.clientY;
    sw = el.offsetWidth; sh = el.offsetHeight;
    document.onmousemove = e => {
      if (!on) return;
      el.style.width  = Math.max(400, sw + e.clientX - sx) + 'px';

    };
    document.onmouseup = () => { on = false; document.onmousemove = null; document.onmouseup = null; };
  };
}

/* ── STATE ──────────────────────────────────────────────────────── */
const WM = {
  wins: {},          // winId → { el, labId, minimized, maximized, savedRect }
  order: [],         // winId[] por foco
  nextId: 1,
  dragSrc: null,     // tile em drag
  ctxEl: null,       // context menu ativo

/* ── REGISTRY (importado do lb-tiles.js via window.LB_LABS) ──── */
function _lab(id) {
  return (window.LB_LABS || []).find(l => l.id === id) || null;
}

/* ── OPEN WINDOW ────────────────────────────────────────────────── */
window.openWindow = function(labId) {
  const lab = _lab(labId);
  if (!lab) { console.warn('[WM] lab não encontrado:', labId); return; }

  // Se já existe, foca ou restaura
  const existing = Object.values(WM.wins).find(w => w.labId === labId);
  if (existing) {
    if (existing.minimized) wmRestore(existing.winId);
    else wmFocus(existing.winId);
    return;
  }

  const winId  = 'w' + (WM.nextId++);
  const mobile = window.innerWidth < 560;
  const off    = ((WM.nextId - 2) % 9) * 22;

  const x = mobile ? 0 : Math.min(40 + off, window.innerWidth  - 460);
  const y = mobile ? 0 : Math.min(50 + off, window.innerHeight - 340);
  const w = mobile ? window.innerWidth      : Math.min(560, window.innerWidth  - x - 16);
  const h = mobile ? window.innerHeight - 50: Math.min(460, window.innerHeight - y - 56);

  const el = document.createElement('div');
  el.className   = 'lb-win is-active';
  el.id          = winId;
  el.dataset.labId = labId;
  el.style.cssText = `left:${x}px;top:${y}px;width:${w}px;height:${h}px;`;

  el.innerHTML = `
    <div class="lb-win-bar" style="background:${lab.cor||'#0078D7'};" data-wid="${winId}">
      <span class="lb-win-bar-icon">${lab.icon}</span>
      <span class="lb-win-bar-title">${lab.nome}</span>
      <div class="lb-win-controls">
        <button class="lb-win-btn" data-action="min"   title="Minimizar">─</button>
        <button class="lb-win-btn" data-action="max"   title="Maximizar">□</button>
        <button class="lb-win-btn" data-action="close" title="Fechar">✕</button>
      </div>
    </div>
    <div class="lb-win-body">
      <iframe class="lb-win-iframe" src="${lab.src}" loading="lazy"
              allow="microphone; camera; fullscreen"
              sandbox="allow-same-origin allow-scripts allow-forms allow-popups allow-downloads">
      </iframe>
    </div>
    <div class="lb-win-resize" data-wid="${winId}"></div>`;

  document.body.appendChild(el);
  WM.wins[winId] = { el, labId, winId, minimized: false, maximized: false };
  WM.order.push(winId);

  _bindDrag(el, winId);
  _bindResize(el.querySelector('.lb-win-resize'), winId);
  _bindButtons(el, winId);
  _addTaskbarPill(winId, lab);
  wmFocus(winId);

  if (window.LBNotif) LBNotif.add(lab.nome, `${lab.nome} aberto`, 'info');
};

/* ── CLOSE ──────────────────────────────────────────────────────── */
window.wmClose = function(winId) {
  const win = WM.wins[winId];
  if (!win) return;
  win.el.style.animation = 'win-open .18s reverse both';
  setTimeout(() => {
    win.el.remove();
    delete WM.wins[winId];
    WM.order = WM.order.filter(id => id !== winId);
    _removeTaskbarPill(winId);
  }, 180);
};

/* ── MINIMIZE ───────────────────────────────────────────────────── */
window.wmMinimize = function(winId) {
  const win = WM.wins[winId];
  if (!win) return;
  win.minimized = true;
  win.el.classList.add('is-minimized');
  _pillSetActive(winId, false);
};

/* ── RESTORE ────────────────────────────────────────────────────── */
window.wmRestore = function(winId) {
  const win = WM.wins[winId];
  if (!win) return;
  win.minimized = false;
  win.el.classList.remove('is-minimized');
  wmFocus(winId);
};

/* ── MAXIMIZE / RESTORE ─────────────────────────────────────────── */
window.wmToggleMax = function(winId) {
  const win = WM.wins[winId];
  if (!win) return;
  if (!win.maximized) {
    win.savedRect = { l: win.el.offsetLeft, t: win.el.offsetTop, w: win.el.offsetWidth, h: win.el.offsetHeight };
    win.maximized = true;
    win.el.classList.add('is-maximized');
  } else {
    win.maximized = false;
    win.el.classList.remove('is-maximized');
    const r = win.savedRect;
    if (r) win.el.style.cssText = `left:${r.l}px;top:${r.t}px;width:${r.w}px;height:${r.h}px;`;
  }
  wmFocus(winId);
};

/* ── FOCUS ──────────────────────────────────────────────────────── */
window.wmFocus = function(winId) {
  Object.values(WM.wins).forEach(w => {
    w.el.classList.remove('is-active');
    w.el.style.zIndex = 100;
    _pillSetActive(w.winId, false);
  });
  const win = WM.wins[winId];
  if (!win) return;
  win.el.classList.add('is-active');
  win.el.style.zIndex = 200;
  WM.order = WM.order.filter(id => id !== winId);
  WM.order.push(winId);
  _pillSetActive(winId, true);
};

/* ── BIND BUTTONS ───────────────────────────────────────────────── */
function _bindButtons(el, winId) {
  el.querySelector('[data-action=min]').addEventListener('click', e => {
    e.stopPropagation(); wmMinimize(winId);
  });
  el.querySelector('[data-action=max]').addEventListener('click', e => {
    e.stopPropagation(); wmToggleMax(winId);
  });
  el.querySelector('[data-action=close]').addEventListener('click', e => {
    e.stopPropagation(); wmClose(winId);
  });
  el.addEventListener('mousedown', () => wmFocus(winId));
}

/* ── DRAG ───────────────────────────────────────────────────────── */
function _bindDrag(el, winId) {
  const bar = el.querySelector('.lb-win-bar');
  let ox = 0, oy = 0, active = false;

  function start(e) {
    if (e.target.closest('.lb-win-controls')) return;
    if (WM.wins[winId]?.maximized) return;
    active = true;
    const p = e.touches ? e.touches[0] : e;
    ox = p.clientX - el.offsetLeft;
    oy = p.clientY - el.offsetTop;
    wmFocus(winId);
    e.preventDefault();
    on('mousemove', move); on('mouseup', end);
    on('touchmove', move, { passive: false }); on('touchend', end);
  }
  function move(e) {
    if (!active) return;
    const p = e.touches ? e.touches[0] : e;
    el.style.left = Math.max(0, Math.min(p.clientX - ox, window.innerWidth  - el.offsetWidth)) + 'px';
    el.style.top  = Math.max(28, Math.min(p.clientY - oy, window.innerHeight - el.offsetHeight - 50)) + 'px';
    if (e.cancelable) e.preventDefault();
  }
  function end() {
    active = false;
    off('mousemove', move); off('mouseup', end);
    off('touchmove', move); off('touchend', end);
  }
  bar.addEventListener('mousedown',  start);
  bar.addEventListener('touchstart', start, { passive: false });
}

/* ── RESIZE ─────────────────────────────────────────────────────── */
function _bindResize(handle, winId) {
  if (!handle) return;
  let sx, sy, sw, sh, active = false;
  function start(e) {
    active = true;
    const p = e.touches ? e.touches[0] : e;
    sx = p.clientX; sy = p.clientY;
    sw = WM.wins[winId].el.offsetWidth;
    sh = WM.wins[winId].el.offsetHeight;
    e.preventDefault();
    on('mousemove', move); on('mouseup', end);
    on('touchmove', move, { passive: false }); on('touchend', end);
  }
  function move(e) {
    if (!active) return;
    const p = e.touches ? e.touches[0] : e;
    const win = WM.wins[winId];
    win.el.style.width  = Math.max(280, sw + p.clientX - sx) + 'px';
    win.el.style.height = Math.max(200, sh + p.clientY - sy) + 'px';
    if (e.cancelable) e.preventDefault();
  }
  function end() {
    active = false;
    off('mousemove', move); off('mouseup', end);
    off('touchmove', move); off('touchend', end);
  }
  handle.addEventListener('mousedown',  start);
  handle.addEventListener('touchstart', start, { passive: false });
}

/* ── TASKBAR PILLS ──────────────────────────────────────────────── */
function _addTaskbarPill(winId, lab) {
  const area = document.getElementById('tb-open-wins');
  if (!area) return;
  const pill = document.createElement('div');
  pill.className = 'tb-win-pill active';
  pill.id = 'pill-' + winId;
  pill.innerHTML = `<span class="tb-win-pill-icon">${lab.icon}</span>${lab.nome.slice(0, 12)}`;
  pill.addEventListener('click', () => {
    const win = WM.wins[winId];
    if (!win) return;
    if (win.minimized)                 wmRestore(winId);
    else if (win.el.classList.contains('is-active')) wmMinimize(winId);
    else                               wmFocus(winId);
  });
  area.appendChild(pill);
}
function _removeTaskbarPill(winId) {
  document.getElementById('pill-' + winId)?.remove();
}
function _pillSetActive(winId, on) {
  document.getElementById('pill-' + winId)?.classList.toggle('active', on);
}

/* ── TILE DRAG & DROP ───────────────────────────────────────────── */
window.initTileDnD = function() {
  document.querySelectorAll('.tile[data-lab-id]').forEach(tile => {
    tile.setAttribute('draggable', 'true');

    tile.addEventListener('dragstart', e => {
      WM.dragSrc = tile;
      e.dataTransfer.effectAllowed = 'move';
      setTimeout(() => tile.classList.add('tile-drag-src'), 0);
    });
    tile.addEventListener('dragend', () => {
      tile.classList.remove('tile-drag-src');
      document.querySelectorAll('.tile-drag-over').forEach(t => t.classList.remove('tile-drag-over'));
      WM.dragSrc = null;
    });
    tile.addEventListener('dragover', e => {
      e.preventDefault(); e.dataTransfer.dropEffect = 'move';
      if (WM.dragSrc && WM.dragSrc !== tile) tile.classList.add('tile-drag-over');
    });
    tile.addEventListener('dragleave', () => tile.classList.remove('tile-drag-over'));
    tile.addEventListener('drop', e => {
      e.preventDefault();
      tile.classList.remove('tile-drag-over');
      if (!WM.dragSrc || WM.dragSrc === tile) return;
      const p = tile.parentNode;
      const si = [...p.children].indexOf(WM.dragSrc);
      const ti = [...p.children].indexOf(tile);
      if (si < ti) p.insertBefore(WM.dragSrc, tile.nextSibling);
      else         p.insertBefore(WM.dragSrc, tile);
      _saveTileOrder();
    });
  });
};

function _saveTileOrder() {
  const order = [...document.querySelectorAll('.tile[data-lab-id]')].map(t => t.dataset.labId);
  try { localStorage.setItem('lb_tile_order', JSON.stringify(order)); } catch(e) {}
}

/* ── KEYBOARD ───────────────────────────────────────────────────── */
window.initKeyboard = function() {
  document.addEventListener('keydown', e => {
    if ((e.ctrlKey || e.metaKey) && e.key === 'f') {
      e.preventDefault(); window.openSearch?.();
    }
    if (e.key === 'Escape') {
      window.closeSearch?.();
      window.closeNotifPanel?.();
    }
    // Alt+F4 fecha janela ativa
    if (e.altKey && e.key === 'F4') {
      const top = WM.order[WM.order.length - 1];
      if (top) wmClose(top);
    }
  });
};

/* ── SCROLL INDICATOR ───────────────────────────────────────────── */
window.initScrollInd = function() {
  const desktop = document.getElementById('lb-desktop');
  const pips    = document.querySelectorAll('.si-pip');
  if (!desktop || !pips.length) return;
  desktop.addEventListener('scroll', () => {
    const pct = desktop.scrollTop / Math.max(desktop.scrollHeight - desktop.clientHeight, 1);
    const idx = Math.min(Math.floor(pct * pips.length), pips.length - 1);
    pips.forEach((p, i) => p.classList.toggle('active', i === idx));
  });
};

/* ── UTILS ──────────────────────────────────────────────────────── */
function on(ev, fn, opts) { document.addEventListener(ev, fn, opts); }
function off(ev, fn)      { document.removeEventListener(ev, fn); }
