function getBase() {
  let base = lsGet(LS_BASE, null);
  if (!base) {
    base = prompt('🔐 Primeiro acesso.\nConfigure sua chave-mestra (parte fixa da senha):\n\nEsta informação não será exibida novamente — anote com segurança:');
    if (!base || !base.trim()) base = 'LAB';
    lsSet(LS_BASE, base.trim());
    alert('✅ Chave configurada.\nSua senha hoje é:\n\n' + buildPass(base.trim()));
  }
  return base;
}
function buildPass(base) {
  const now = new Date();
  const dd = String(now.getDate()).padStart(2,'0');
  const mm = String(now.getMonth()+1).padStart(2,'0');
  return base + dd + mm + '123';
}

  const now = new Date();

}

function loadAll() {
  const stored = lsGet(LS_ASSETS, null);
  // Merge: stored + any seed assets not yet in storage
  if (stored && stored.length) {
    const storedIds = stored.map(a=>a.id);
    const newSeeds = SEED_ASSETS.filter(s=>!storedIds.includes(s.id));
    STATE.assets = [...stored, ...newSeeds];
  } else {
    STATE.assets = [...SEED_ASSETS];
  }
  saveAll();
}

function onImportJson(evt){
  const file=evt.target.files[0]; if(!file)return;
  const r=new FileReader();
  r.onload=e=>{
    try{
      const data=JSON.parse(e.target.result);
      const arr=Array.isArray(data)?data:(data.assets||[]);
      if(!arr.length){toast('JSON vazio.','red');return;}
      const existing=STATE.assets.map(a=>a.id);
      const news=arr.filter(a=>!existing.includes(a.id));
      STATE.assets=[...STATE.assets,...news];
      saveAll();renderAdminList();renderAdminChips();
      toast(`${news.length} novo(s) protocolo(s) importado(s) ✓`,'gold');
    }catch{toast('Erro ao ler JSON.','red');}
  };
  r.readAsText(file); evt.target.value='';
}
function onImportAsset(evt){
  const file=evt.target.files[0]; if(!file)return;
  const r=new FileReader();
  r.onload=e=>{
    const id='LB-IMP-'+genId().slice(0,4);
    const newA={id,titulo:file.name.replace(/\.[^.]+$/,''),versao:'v1.0',categoria:'Código',status:'draft',tags:['Importado'],descricao:'Ativo importado de arquivo HTML.',demoUrl:'',html:e.target.result,css:'',js:'',docs:'',legal:'',criado:new Date().toISOString().slice(0,10)};
    STATE.assets.push(newA); saveAll(); renderAdminList(); renderAdminChips();
    toast('"'+newA.titulo+'" importado ✓','cyber');
  };
  r.readAsText(file); evt.target.value='';
}
